📄 ${file.name}
`).join(''); step2.style.display = 'block'; } else { fileCount.innerHTML = 'Belum ada foto yang dipilih.'; fileList.style.display = 'none'; step2.style.display = 'none'; } } function logProgress(msg) { const logBox = document.getElementById('progressLog'); logBox.innerHTML += `
> ${msg}
`; logBox.scrollTop = logBox.scrollHeight; } window.startProcessing = async function() { if (!window.globalSelectedFiles || window.globalSelectedFiles.length === 0) return; document.getElementById('step1').style.display = 'none'; document.getElementById('step2').style.display = 'none'; const progressContainer = document.getElementById('progressContainer'); progressContainer.style.display = 'block'; const statusTitle = document.getElementById('statusTitle'); const previewArea = document.getElementById('previewArea'); document.getElementById('progressLog').innerHTML = ''; try { if (!segmenter) { logProgress("Mengunduh model AI (MODNet)..."); logProgress("Proses ini memakan data ~25MB, harap tunggu..."); // PERBAIKAN: Mengganti model RMBG-1.4 yang tidak disupport ke Xenova/modnet segmenter = await pipeline('image-segmentation', 'Xenova/modnet', { progress_callback: (info) => { if (info.status === 'progress' && info.name) { let percent = Math.round((info.loaded / info.total) * 100); if(percent % 20 === 0) { logProgress(`Download ${info.name}: ${percent}%`); } } else if (info.status === 'done') { logProgress(`Selesai download ${info.file}`); } } }); logProgress("Model AI berhasil dimuat!"); } for (let i = 0; i < window.globalSelectedFiles.length; i++) { const file = window.globalSelectedFiles[i]; logProgress(`Memproses foto (${i+1}/${window.globalSelectedFiles.length}): ${file.name}`); try { const originalImageUrl = URL.createObjectURL(file); logProgress("Sedang memotong background..."); // Eksekusi AI MODNet const result = await segmenter(originalImageUrl); // Membuat gambar PNG transparan const canvas = document.createElement('canvas'); const ctx = canvas.getContext('2d'); const img = new Image(); img.src = originalImageUrl; await new Promise((resolve, reject) => { img.onload = () => { canvas.width = img.width; canvas.height = img.height; // 1. ${file.name}

Asli

Hasil

⬇️ Unduh Gambar Ini `; document.getElementById('previewArea').appendChild(previewCard); logProgress(`✅ Sukses: ${file.name}`); resolve(); }, 'image/png'); }; // Ekstrak URL data dari RawImage object Transformers.js maskRawImage.toDataURL().then(url => { maskImg.src = url; }); }; }); } catch (err) { logProgress(`❌ Gagal foto ${file.name}: ${err.message}`); } } statusTitle.innerHTML = `🎉 Semua proses selesai!`; } catch (globalErr) { logProgress(`❌ ERROR FATAL: ${globalErr.message}`); statusTitle.innerHTML = `Gagal memuat AI.
